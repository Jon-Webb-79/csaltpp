1930  throw std::invalid_argument("Only square matrices can be inverted");
1931

1933  DenseMatrix<T> A(*this);
1934  DenseMatrix<T> I(n, n, T{});
1935  for (std::size_t i = 0; i < n; ++i)
1936  I.update(i, i, T{1}); // Identity matrix
1937 
1938  for (std::size_t i = 0; i < n; ++i) {
1939  // Pivot selection (partial pivoting)
1940  std::size_t pivot = i;
1941  T max_val = std::abs(A.get(i, i));
1942  for (std::size_t j = i + 1; j < n; ++j) {
1943  T val = std::abs(A.get(j, i));
1944  if (val > max_val) {
1945  max_val = val;
1946  pivot = j;
1947  }
1948  }
1949 
1950  if (max_val == T{})
1951  throw std::runtime_error("Matrix is singular and cannot be inverted");
1952 
1953  // Swap rows (corrected)
1954  if (pivot != i) {
1955  for (std::size_t k = 0; k < n; ++k) {
1956  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1957  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1958  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1959  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1960  }
1961  }
1962

1964  T pivot_val = A.data[i * n + i];
1965  for (std::size_t k = 0; k < n; ++k) {
1966  A.data[i * n + k] /= pivot_val;
1967  A.init[i * n + k] = 1;
1968

1970  I.init[i * n + k] = 1;
1971  }

1974  for (std::size_t j = 0; j < n; ++j) {
1975  if (j == i) continue;
1976  T factor = A.data[j * n + i];
1977  for (std::size_t k = 0; k < n; ++k) {
1978  A.data[j * n + k] -= factor * A.data[i * n + k];
1979  A.init[j * n + k] = 1;
