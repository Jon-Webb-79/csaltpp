1737  DenseMatrix<T> inverse() const {
1738  if (rows_ != cols_)
1739  throw std::invalid_argument("Only square matrices can be inverted");
1740 
1741  const std::size_t n = rows_;
1742  DenseMatrix<T> A(*this);
1743  DenseMatrix<T> I(n, n, T{});
1744  for (std::size_t i = 0; i < n; ++i)
1745  I.update(i, i, T{1}); // Identity matrix
1746 
1747  for (std::size_t i = 0; i < n; ++i) {
1748  // Pivot selection (partial pivoting)
1749  std::size_t pivot = i;
1750  T max_val = std::abs(A.get(i, i));
1751  for (std::size_t j = i + 1; j < n; ++j) {
1752  T val = std::abs(A.get(j, i));
1753  if (val > max_val) {
1754  max_val = val;
1755  pivot = j;
1756  }
1757  }
1758 
1759  if (max_val == T{})
1760  throw std::runtime_error("Matrix is singular and cannot be inverted");
1761 
1762  // Swap rows (corrected)
1763  if (pivot != i) {
1764  for (std::size_t k = 0; k < n; ++k) {
1765  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1766  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1767  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1768  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1769  }
1770  }
1771 
1772  // Normalize pivot row
1773  T pivot_val = A.data[i * n + i];
1774  for (std::size_t k = 0; k < n; ++k) {
1775  A.data[i * n + k] /= pivot_val;
1776  A.init[i * n + k] = 1;
1777 
1778  I.data[i * n + k] /= pivot_val;
1779  I.init[i * n + k] = 1;
1780  }
1781 
1782  // Eliminate other rows
1783  for (std::size_t j = 0; j < n; ++j) {
1784  if (j == i) continue;
1785  T factor = A.data[j * n + i];
1786  for (std::size_t k = 0; k < n; ++k) {
1787  A.data[j * n + k] -= factor * A.data[i * n + k];
1788  A.init[j * n + k] = 1;
1789 
1790  I.data[j * n + k] -= factor * I.data[i * n + k];
1791  I.init[j * n + k] = 1;
1792  }
1793  }
1794  }
1795 
1796  return I;
1797  }
