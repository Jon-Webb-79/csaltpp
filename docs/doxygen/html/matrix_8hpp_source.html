1708  DenseMatrix<T> inverse() const {
1709  if (rows_ != cols_)
1710  throw std::invalid_argument("Only square matrices can be inverted");
1711 
1712  const std::size_t n = rows_;
1713  DenseMatrix<T> A(*this);
1714  DenseMatrix<T> I(n, n, T{});
1715  for (std::size_t i = 0; i < n; ++i)
1716  I.update(i, i, T{1}); // Identity matrix
1717 
1718  for (std::size_t i = 0; i < n; ++i) {
1719  // Pivot selection (partial pivoting)
1720  std::size_t pivot = i;
1721  T max_val = std::abs(A.get(i, i));
1722  for (std::size_t j = i + 1; j < n; ++j) {
1723  T val = std::abs(A.get(j, i));
1724  if (val > max_val) {
1725  max_val = val;
1726  pivot = j;
1727  }
1728  }
1729 
1730  if (max_val == T{})
1731  throw std::runtime_error("Matrix is singular and cannot be inverted");
1732 
1733  // Swap rows (corrected)
1734  if (pivot != i) {
1735  for (std::size_t k = 0; k < n; ++k) {
1736  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1737  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1738  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1739  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1740  }
1741  }
1742 
1743  // Normalize pivot row
1744  T pivot_val = A.data[i * n + i];
1745  for (std::size_t k = 0; k < n; ++k) {
1746  A.data[i * n + k] /= pivot_val;
1747  A.init[i * n + k] = 1;
1748 
1749  I.data[i * n + k] /= pivot_val;
1750  I.init[i * n + k] = 1;
1751  }
1752 
1753  // Eliminate other rows
1754  for (std::size_t j = 0; j < n; ++j) {
1755  if (j == i) continue;
1756  T factor = A.data[j * n + i];
1757  for (std::size_t k = 0; k < n; ++k) {
1758  A.data[j * n + k] -= factor * A.data[i * n + k];
1759  A.init[j * n + k] = 1;
1760 
1761  I.data[j * n + k] -= factor * I.data[i * n + k];
1762  I.init[j * n + k] = 1;
1763  }
1764  }
1765  }
1766 
1767  return I;
1768  }
