1824  DenseMatrix<T> inverse() const {
1825  if (rows_ != cols_)
1826  throw std::invalid_argument("Only square matrices can be inverted");
1827 
1828  const std::size_t n = rows_;
1829  DenseMatrix<T> A(*this);
1830  DenseMatrix<T> I(n, n, T{});
1831  for (std::size_t i = 0; i < n; ++i)
1832  I.update(i, i, T{1}); // Identity matrix
1833 
1834  for (std::size_t i = 0; i < n; ++i) {
1835  // Pivot selection (partial pivoting)
1836  std::size_t pivot = i;
1837  T max_val = std::abs(A.get(i, i));
1838  for (std::size_t j = i + 1; j < n; ++j) {
1839  T val = std::abs(A.get(j, i));
1840  if (val > max_val) {
1841  max_val = val;
1842  pivot = j;
1843  }
1844  }
1845 
1846  if (max_val == T{})
1847  throw std::runtime_error("Matrix is singular and cannot be inverted");
1848 
1849  // Swap rows (corrected)
1850  if (pivot != i) {
1851  for (std::size_t k = 0; k < n; ++k) {
1852  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1853  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1854  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1855  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1856  }
1857  }
1858 
1859  // Normalize pivot row
1860  T pivot_val = A.data[i * n + i];
1861  for (std::size_t k = 0; k < n; ++k) {
1862  A.data[i * n + k] /= pivot_val;
1863  A.init[i * n + k] = 1;
1864 
1865  I.data[i * n + k] /= pivot_val;
1866  I.init[i * n + k] = 1;
1867  }
1868 
1869  // Eliminate other rows
1870  for (std::size_t j = 0; j < n; ++j) {
1871  if (j == i) continue;
1872  T factor = A.data[j * n + i];
1873  for (std::size_t k = 0; k < n; ++k) {
1874  A.data[j * n + k] -= factor * A.data[i * n + k];
1875  A.init[j * n + k] = 1;
1876 
1877  I.data[j * n + k] -= factor * I.data[i * n + k];
1878  I.init[j * n + k] = 1;
1879  }
1880  }
1881  }
1882 
1883  return I;
1884  }
