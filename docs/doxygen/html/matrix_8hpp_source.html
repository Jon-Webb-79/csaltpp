1730  DenseMatrix<T> inverse() const {
1731  if (rows_ != cols_)
1732  throw std::invalid_argument("Only square matrices can be inverted");
1733 
1734  const std::size_t n = rows_;
1735  DenseMatrix<T> A(*this);
1736  DenseMatrix<T> I(n, n, T{});
1737  for (std::size_t i = 0; i < n; ++i)
1738  I.update(i, i, T{1}); // Identity matrix
1739 
1740  for (std::size_t i = 0; i < n; ++i) {
1741  // Pivot selection (partial pivoting)
1742  std::size_t pivot = i;
1743  T max_val = std::abs(A.get(i, i));
1744  for (std::size_t j = i + 1; j < n; ++j) {
1745  T val = std::abs(A.get(j, i));
1746  if (val > max_val) {
1747  max_val = val;
1748  pivot = j;
1749  }
1750  }
1751 
1752  if (max_val == T{})
1753  throw std::runtime_error("Matrix is singular and cannot be inverted");
1754 
1755  // Swap rows (corrected)
1756  if (pivot != i) {
1757  for (std::size_t k = 0; k < n; ++k) {
1758  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1759  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1760  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1761  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1762  }
1763  }
1764 
1765  // Normalize pivot row
1766  T pivot_val = A.data[i * n + i];
1767  for (std::size_t k = 0; k < n; ++k) {
1768  A.data[i * n + k] /= pivot_val;
1769  A.init[i * n + k] = 1;
1770 
1771  I.data[i * n + k] /= pivot_val;
1772  I.init[i * n + k] = 1;
1773  }
1774 
1775  // Eliminate other rows
1776  for (std::size_t j = 0; j < n; ++j) {
1777  if (j == i) continue;
1778  T factor = A.data[j * n + i];
1779  for (std::size_t k = 0; k < n; ++k) {
1780  A.data[j * n + k] -= factor * A.data[i * n + k];
1781  A.init[j * n + k] = 1;
1782 
1783  I.data[j * n + k] -= factor * I.data[i * n + k];
1784  I.init[j * n + k] = 1;
1785  }
1786  }
1787  }
1788 
1789  return I;
1790  }
