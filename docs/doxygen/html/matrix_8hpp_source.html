1835

1838  DenseMatrix<T> I(n, n, T{});
1839  for (std::size_t i = 0; i < n; ++i)
1840  I.update(i, i, T{1}); // Identity matrix
1841 
1842  for (std::size_t i = 0; i < n; ++i) {
1843  // Pivot selection (partial pivoting)
1844  std::size_t pivot = i;
1845  T max_val = std::abs(A.get(i, i));
1846  for (std::size_t j = i + 1; j < n; ++j) {
1847  T val = std::abs(A.get(j, i));
1848  if (val > max_val) {
1849  max_val = val;
1850  pivot = j;
1851  }
1852  }
1853

1856 
1857  // Swap rows (corrected)
1858  if (pivot != i) {
1859  for (std::size_t k = 0; k < n; ++k) {
1860  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1861  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
