1863  throw std::invalid_argument("Only square matrices can be inverted");

1867  DenseMatrix<T> I(n, n, T{});
1868  for (std::size_t i = 0; i < n; ++i)
1869  I.update(i, i, T{1}); // Identity matrix
1870 
1871  for (std::size_t i = 0; i < n; ++i) {
1872  // Pivot selection (partial pivoting)
1873  std::size_t pivot = i;
1874  T max_val = std::abs(A.get(i, i));
1875  for (std::size_t j = i + 1; j < n; ++j) {
1876  T val = std::abs(A.get(j, i));
1877  if (val > max_val) {
1878  max_val = val;
1879  pivot = j;
1880  }
1881  }
1882 
1883  if (max_val == T{})
1884  throw std::runtime_error("Matrix is singular and cannot be inverted");
1885 
1886  // Swap rows (corrected)
1887  if (pivot != i) {
1888  for (std::size_t k = 0; k < n; ++k) {
1889  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1890  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1891  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1892  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1893  }
1894  }
1895

1897  T pivot_val = A.data[i * n + i];
1898  for (std::size_t k = 0; k < n; ++k) {
1899  A.data[i * n + k] /= pivot_val;
1900  A.init[i * n + k] = 1;
1901

1903  I.init[i * n + k] = 1;
1904  }
1905

1907  for (std::size_t j = 0; j < n; ++j) {
1908  if (j == i) continue;

1910  for (std::size_t k = 0; k < n; ++k) {
