1818  DenseMatrix<T> inverse() const {
1819  if (rows_ != cols_)
1820  throw std::invalid_argument("Only square matrices can be inverted");
1821 
1822  const std::size_t n = rows_;
1823  DenseMatrix<T> A(*this);
1824  DenseMatrix<T> I(n, n, T{});
1825  for (std::size_t i = 0; i < n; ++i)
1826  I.update(i, i, T{1}); // Identity matrix
1827 
1828  for (std::size_t i = 0; i < n; ++i) {
1829  // Pivot selection (partial pivoting)
1830  std::size_t pivot = i;
1831  T max_val = std::abs(A.get(i, i));
1832  for (std::size_t j = i + 1; j < n; ++j) {
1833  T val = std::abs(A.get(j, i));
1834  if (val > max_val) {
1835  max_val = val;
1836  pivot = j;
1837  }
1838  }
1839 
1840  if (max_val == T{})
1841  throw std::runtime_error("Matrix is singular and cannot be inverted");
1842 
1843  // Swap rows (corrected)
1844  if (pivot != i) {
1845  for (std::size_t k = 0; k < n; ++k) {
1846  std::swap(A.data[i * n + k], A.data[pivot * n + k]);
1847  std::swap(A.init[i * n + k], A.init[pivot * n + k]);
1848  std::swap(I.data[i * n + k], I.data[pivot * n + k]);
1849  std::swap(I.init[i * n + k], I.init[pivot * n + k]);
1850  }
1851  }
1852 
1853  // Normalize pivot row
1854  T pivot_val = A.data[i * n + i];
1855  for (std::size_t k = 0; k < n; ++k) {
1856  A.data[i * n + k] /= pivot_val;
1857  A.init[i * n + k] = 1;
1858 
1859  I.data[i * n + k] /= pivot_val;
1860  I.init[i * n + k] = 1;
1861  }
1862 
1863  // Eliminate other rows
1864  for (std::size_t j = 0; j < n; ++j) {
1865  if (j == i) continue;
1866  T factor = A.data[j * n + i];
1867  for (std::size_t k = 0; k < n; ++k) {
1868  A.data[j * n + k] -= factor * A.data[i * n + k];
1869  A.init[j * n + k] = 1;
1870 
1871  I.data[j * n + k] -= factor * I.data[i * n + k];
1872  I.init[j * n + k] = 1;
1873  }
1874  }
1875  }
1876 
1877  return I;
1878  }
